156 explicit constexpr Set(SourceRange&& source) noexcept(

158 static_assert(concepts::NothrowRangeConstruction<value_type, SourceRange>,

255 if (idx < result.logical_size_) {
256 if (idx + 1U < result.logical_size_) {
