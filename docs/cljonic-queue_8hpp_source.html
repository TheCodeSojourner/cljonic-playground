160 template <std::ranges::input_range SourceRange>
161 requires(!concepts_detail::is_cljonic_collection_v<SourceRange> &&
162 !concepts_detail::is_cljonic_producer_v<SourceRange> &&
163 !std::same_as<std::remove_cvref_t<SourceRange>, value_type>)
164 explicit constexpr Queue(SourceRange&& source) noexcept(
165 (concepts::NothrowRangeConstruction<value_type, SourceRange>)) {
166 static_assert(
167 concepts::NothrowRangeConstruction<value_type, SourceRange>,
168 "cljonic::Queue: range/view constructor requires safe element conversion and non-throwing traversal");
169 static_assert(concepts_detail::static_extent_fits_v<SourceRange, CapacityValue>,
170 "cljonic::Queue: static-extent range source exceeds Queue CapacityValue");
171
172 std::size_t copy_count = 0;
173 for (auto&& item : std::forward<SourceRange>(source)) {
174 if (copy_count++ >= CapacityValue) {
175 break;
176 }
177 *this = conj(value_type(std::forward<decltype(item)>(item)));
178 }
179 }

236 [[nodiscard]] constexpr auto conj(const T& element) const noexcept -> Queue {
237 Queue result = *this;
238 if constexpr (CapacityValue > 0U) {
239 if (result.logical_size_ < CapacityValue) {
240 const auto tail = (result.head_ + result.logical_size_) % CapacityValue;
241 result.elements_[tail] = element;
242 ++result.logical_size_;
243 }
244 }
245 return result;
246 }

250 [[nodiscard]] constexpr auto peek() const noexcept -> T {
251 return (logical_size_ > 0) ? elements_[head_] : T{};
252 }

256 [[nodiscard]] constexpr auto pop() const noexcept -> Queue {
257 Queue result = *this;
258 if constexpr (CapacityValue > 0U) {
259 if (result.logical_size_ > 0) {
260 result.head_ = (result.head_ + 1) % CapacityValue;
261 --result.logical_size_;
262 }
263 }
264 return result;
