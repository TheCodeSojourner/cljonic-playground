139 constexpr Map(SourceRange&& source) noexcept(

141 using source_value_type = std::ranges::range_value_t<SourceRange>;
142 static_assert(concepts::NothrowElementConstruction<value_type, source_value_type>,

145 static_assert(concepts_detail::static_extent_fits_v<SourceRange, CapacityValue>,

213 Map result = *this;
