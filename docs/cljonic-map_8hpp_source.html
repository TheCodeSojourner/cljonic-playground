140 template <std::ranges::input_range SourceRange>

142 !concepts_detail::is_cljonic_producer_v<SourceRange> &&

144 constexpr Map(SourceRange&& source) noexcept((concepts::NothrowRangeConstruction<value_type, SourceRange>)) {
145 static_assert(concepts::NothrowRangeConstruction<value_type, SourceRange>,
146 "cljonic::Map: range/view constructor requires each source element to construct "
147 "the exact MapEntry type and requires non-throwing traversal");
148 static_assert(concepts_detail::static_extent_fits_v<SourceRange, CapacityValue>,
149 "cljonic::Map: static-extent range source exceeds Map CapacityValue");

153 if (copy_count++ >= CapacityValue) {

156 *this = assoc_entry(value_type{std::forward<decltype(item)>(item)});
