180 // A cljonic collection or producer is not a SourceConstruction source
181 // (REQ-FN-027A): it is admissible only as one element of exactly its own
182 // type (the pack constructor) or enclosed via EnclosureConstruction;
183 // materializing it is the role of into, preflighted by fits_into. A lone
184 // argument that is exactly ElementType likewise stays with the pack
185 // constructor.
186 template <std::ranges::input_range SourceRange>
187 requires(!concepts_detail::is_cljonic_collection_v<SourceRange> &&
188 !concepts_detail::is_cljonic_producer_v<SourceRange> &&
189 !std::same_as<std::remove_cvref_t<SourceRange>, ElementType>)
190 constexpr Vector(SourceRange&& source) noexcept((concepts::NothrowRangeConstruction<ElementType, SourceRange>)) {
191 static_assert(concepts::NothrowRangeConstruction<ElementType, SourceRange>,
192 "cljonic::Vector: range/view constructor requires each source element to "
193 "convert safely to ElementType and requires non-throwing traversal");
194 static_assert(concepts_detail::static_extent_fits_v<SourceRange, CapacityValue>,
195 "cljonic::Vector: static-extent range source exceeds Vector CapacityValue");
196
197 std::size_t copy_count = 0;
198 for (auto&& item : std::forward<SourceRange>(source)) {
199 if (copy_count >= CapacityValue) {
200 break;
201 }
202 storage_[copy_count++] = value_type(std::forward<decltype(item)>(item));
203 }
204 logical_size_ = copy_count;
205 }

207 // Diagnostic fallback (REQ-DIAG-010): a cljonic collection or producer is
208 // not a SourceConstruction source. This overload exists only to explain the
209 // rejection -- it produces no value and is never a supported call target.
210 template <typename SourceValue>
211 requires((concepts_detail::is_cljonic_collection_v<SourceValue> ||
212 concepts_detail::is_cljonic_producer_v<SourceValue>) &&
213 !std::same_as<std::remove_cvref_t<SourceValue>, value_type>)
214 constexpr Vector([[maybe_unused]] SourceValue&& source) noexcept {
215 static_assert(
216 concepts_detail::dependent_false<SourceValue>,
217 "cljonic::Vector: you cannot construct a Vector from a cljonic collection or producer. A Vector is built "
218 "from its individual elements, or from a non-cljonic C++ range or view. To copy this argument's contents "
219 "into a Vector, use into(destination, source) (call fits_into(destination, source) first to check whether "
220 "the whole source fits); to keep it as a single element, write Vector{...}.");
221 }
