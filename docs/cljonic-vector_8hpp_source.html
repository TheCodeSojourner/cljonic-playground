184 // A cljonic collection or producer is not a SourceConstruction source
185 // (REQ-FN-027A): it is admissible only as one element of exactly its own
186 // type (the pack constructor) or enclosed via EnclosureConstruction;
187 // materializing it is the role of into, preflighted by fits_into. A lone
188 // argument that is exactly ElementType likewise stays with the pack
189 // constructor.
190 template <std::ranges::input_range SourceRange>
191 requires(!concepts_detail::is_cljonic_collection_v<SourceRange> &&
192 !concepts_detail::is_cljonic_producer_v<SourceRange> &&
193 !std::same_as<std::remove_cvref_t<SourceRange>, ElementType>)
194 explicit constexpr Vector(SourceRange&& source) noexcept(
195 (concepts::NothrowRangeConstruction<ElementType, SourceRange>)) {
196 static_assert(concepts::NothrowRangeConstruction<ElementType, SourceRange>,
197 "cljonic::Vector: range/view constructor requires each source element to "
198 "convert safely to ElementType and requires non-throwing traversal");
199 static_assert(concepts_detail::static_extent_fits_v<SourceRange, CapacityValue>,
200 "cljonic::Vector: static-extent range source exceeds Vector CapacityValue");
201
202 std::size_t copy_count = 0;
203 for (auto&& item : std::forward<SourceRange>(source)) {
204 if (copy_count >= CapacityValue) {
205 break;
206 }
207 storage_[copy_count++] = value_type(std::forward<decltype(item)>(item));
208 }
209 logical_size_ = copy_count;
210 }

212 // Diagnostic fallback (REQ-DIAG-010): a cljonic collection or producer is
213 // not a SourceConstruction source. This overload exists only to explain the
214 // rejection -- it produces no value and is never a supported call target.
215 template <typename SourceValue>
216 requires((concepts_detail::is_cljonic_collection_v<SourceValue> ||
217 concepts_detail::is_cljonic_producer_v<SourceValue>) &&
218 !std::same_as<std::remove_cvref_t<SourceValue>, value_type>)
219 explicit constexpr Vector([[maybe_unused]] SourceValue&& source) noexcept {
220 static_assert(
221 concepts_detail::dependent_false<SourceValue>,
222 "cljonic::Vector: you cannot construct a Vector from a cljonic collection or producer. A Vector is built "

224 "collection or "
225 "producer into a Vector, use into(destination, source); call fits_into(destination, source) first to "
226 "check whether the complete source fits in the Vector's remaining capacity; to keep it as a single "
227 "element, write Vector{...}.");
