56struct RepeatNextAccess {

58 [[nodiscard]] static constexpr auto tail(const P& producer) noexcept -> P {
59 if (producer.is_finite_) {
60 const auto remaining_count = producer.count_ == 0U ? 0U : producer.count_ - 1U;

67struct RangeNextAccess {
68 template <concepts::CljonicRange P>
69 [[nodiscard]] static constexpr auto tail(const P& producer) noexcept -> P {
70 if (producer.step_ == 0) {
71 return producer;
72 }
73
74 const auto extent = producer.nonzero_step_extent();
75 if (extent == 0U) {
76 return producer;

78 if (extent == 1U) {

345template <concepts_detail::NextSequenceableCollection C>
346[[nodiscard]] constexpr auto next(const C& collection) noexcept -> concepts_detail::NextResult<C>::type {
347 using Result = concepts_detail::NextResult<C>::type;
348 Result result{};
349 auto iterator = collection.begin();
350 const auto end = collection.end();
351 if (iterator != end) {
352 ++iterator;
353 }
354 for (; iterator != end; ++iterator) {
355 result = result.conj(*iterator);
356 }
357 return result;
358}
