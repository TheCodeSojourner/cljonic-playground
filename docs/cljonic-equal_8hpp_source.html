23// The non-cljonic fallthrough domain of `equal` (REQ-FN-002G) is the closed
24// value domain: scalars and scoped enums, aggregate-like structs, and std::
25// variant composites. Standard-library range and container types belong to the
26// C++ interoperability surface, not the equality domain, and are rejected.
27template <typename T>
28inline constexpr bool in_non_cljonic_fallthrough_domain_v =
29 !is_cljonic_collection_v<T> && !is_cljonic_producer_v<T> && !std::ranges::range<T> &&
30 (std::is_scalar_v<T> || std::is_aggregate_v<T> || is_std_variant_v<T>);

32// Equality-family classification for the general-equality free function `equal`
33// (REQ-FN-002G): Vector and Queue join all five producers in the sequential
34// family; Map, Set, and String each form their own family; every other type is
35// outside the family classification.
36template <typename T>
37inline constexpr equality_family equal_family_of_v = []() constexpr -> equality_family {
38 if constexpr (is_cljonic_collection_v<T>) {
39 switch (collection_kind_of_v<T>) {
40 case collection_kind::vector:
41 case collection_kind::queue:
42 return equality_family::sequential;
43 case collection_kind::map:
44 return equality_family::map;
45 case collection_kind::set:
46 return equality_family::set;
47 case collection_kind::string:
48 return equality_family::string;
49 default:
50 return equality_family::none;
51 }
52 } else if constexpr (is_cljonic_producer_v<T>) {
53 return equality_family::sequential;
54 } else {
55 return equality_family::none;
56 }
57}();

102// Compile-time admissibility of one operand pair (REQ-FN-002G): the
103// disjunction of the binary equal overload gates, kept in lockstep with the
104// five binary overloads so every arity rejects the same pairs at compile
105// time. Both members of a family pair always expose the corresponding member
106// types, so the void fallback only guards the non-family substitution path.
107template <typename Lhs, typename Rhs>
108inline constexpr bool equal_pair_admissible_v =
109 (equal_family_of_v<Lhs> == equality_family::none && equal_family_of_v<Rhs> == equality_family::none &&
110 in_non_cljonic_fallthrough_domain_v<Lhs> && in_non_cljonic_fallthrough_domain_v<Rhs> && std::same_as<Lhs, Rhs> &&
111 concepts::StableEqualityComparable<Lhs>) ||
112 (equal_family_of_v<Lhs> == equality_family::sequential && equal_family_of_v<Rhs> == equality_family::sequential &&
113 std::same_as<equal_value_type_of_t<Lhs>, equal_value_type_of_t<Rhs>> &&
114 concepts::StableEqualityComparable<equal_value_type_of_t<Lhs>>) ||
115 (equal_family_of_v<Lhs> == equality_family::string && equal_family_of_v<Rhs> == equality_family::string) ||
116 (equal_family_of_v<Lhs> == equality_family::map && equal_family_of_v<Rhs> == equality_family::map &&
117 std::same_as<equal_key_type_of_t<Lhs>, equal_key_type_of_t<Rhs>> &&
118 std::same_as<equal_association_value_type_of_t<Lhs>, equal_association_value_type_of_t<Rhs>> &&
119 concepts::StableEqualityComparable<equal_key_type_of_t<Lhs>> &&
120 concepts::StableEqualityComparable<equal_association_value_type_of_t<Lhs>>) ||
121 (equal_family_of_v<Lhs> == equality_family::set && equal_family_of_v<Rhs> == equality_family::set &&
122 std::same_as<equal_value_type_of_t<Lhs>, equal_value_type_of_t<Rhs>> &&
123 concepts::StableEqualityComparable<equal_value_type_of_t<Lhs>>);

138// Lazy element-wise bounded-prefix walk: terminates at the first differing
139// element pair and requires both operands to be exhausted simultaneously for
140// equality. All producers expose cap-bounded begin()/end() traversal, so the
141// walk always terminates, including for two unbounded producers.
142namespace equal_walk_detail {
143
144// Reports whether both iterators have reached their respective ends after the
145// walk, which requires the two sequences to have exhausted simultaneously.
146template <typename LhsIterator, typename RhsIterator, typename LhsEnd, typename RhsEnd>
147[[nodiscard]] constexpr auto both_exhausted(const LhsIterator& lhs_iterator, const LhsEnd& lhs_end,
148 const RhsIterator& rhs_iterator, const RhsEnd& rhs_end) noexcept -> bool {
149 return (lhs_iterator == lhs_end) && (rhs_iterator == rhs_end);
150}
151
152} // namespace equal_walk_detail
153
154template <typename Lhs, typename Rhs>
155[[nodiscard]] constexpr auto equal_prefix_walk(const Lhs& lhs, const Rhs& rhs) noexcept -> bool {
156 using equal_walk_detail::both_exhausted;
157 auto lhs_iterator = lhs.begin();
158 const auto lhs_end = lhs.end();
159 auto rhs_iterator = rhs.begin();
160 const auto rhs_end = rhs.end();
161 while ((lhs_iterator != lhs_end) && (rhs_iterator != rhs_end)) {
162 if (!(*lhs_iterator == *rhs_iterator)) {
163 return false;
164 }
165 ++lhs_iterator;
166 ++rhs_iterator;
167 }
168 return both_exhausted(lhs_iterator, lhs_end, rhs_iterator, rhs_end);
169}

175// Order-insensitive Map matching: every logical entry of the left operand must
176// have a matching key in the right operand whose associated value compares
177// equal (REQ-COLL-021 Map equality semantics).
178template <typename Lhs, typename Rhs>
179[[nodiscard]] constexpr auto map_entries_all_match(const Lhs& lhs, const Rhs& rhs) noexcept -> bool {
180 for (const auto& entry : lhs) {
181 if (!rhs.contains(entry.key) || !(rhs(entry.key) == entry.value)) {
182 return false;
183 }
184 }
185 return true;
186}
187
188// Order-insensitive Set matching: every logical element of the left operand
189// must be present in the right operand (REQ-COLL-021 Set equality semantics).
190template <typename Lhs, typename Rhs>
191[[nodiscard]] constexpr auto set_elements_all_contained(const Lhs& lhs, const Rhs& rhs) noexcept -> bool {
192 for (const auto& element : lhs) {
193 if (!rhs.contains(element)) {
194 return false;
195 }
196 }
197 return true;
198}
