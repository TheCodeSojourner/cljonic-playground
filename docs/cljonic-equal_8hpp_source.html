11// The non-cljonic fallthrough domain of `equal` (REQ-FN-002G) is the closed
12// value domain: arithmetic scalars and scoped enums, aggregate-like structs,
13// and cljonic::Variant composites. Standard-library range and container types
14// belong to the C++ interoperability surface, not the equality domain, and are
15// rejected, as is the standard-library variant (it is not a cljonic value
16// type). Unscoped enums and pointers (including member pointers) are also
17// outside the domain: `std::is_scalar_v` would otherwise admit them.
18template <typename T>
19inline constexpr bool in_non_cljonic_fallthrough_domain_v =
20 !is_cljonic_collection_v<T> && !is_cljonic_producer_v<T> && !std::ranges::range<T> &&
21 ((std::is_arithmetic_v<T> || std::is_scoped_enum_v<T>) || std::is_aggregate_v<T> || is_cljonic_variant_v<T>);
22
23// Equality-family classification for the general-equality free function `equal`
24// (REQ-FN-002G): Vector and Queue join all five producers in the sequential
25// family; Map, Set, and String each form their own family; every other type is
26// outside the family classification.
27template <typename T>
28inline constexpr equality_family equal_family_of_v = []() constexpr -> equality_family {
29 if constexpr (is_cljonic_collection_v<T>) {
30 switch (collection_kind_of_v<T>) {
31 case collection_kind::vector:
32 case collection_kind::queue:
33 return equality_family::sequential;
34 case collection_kind::map:
35 return equality_family::map;
36 case collection_kind::set:
37 return equality_family::set;
38 case collection_kind::string:
39 return equality_family::string;
40 default:
41 return equality_family::none;
42 }
43 } else if constexpr (is_cljonic_producer_v<T>) {
44 return equality_family::sequential;
45 } else {
46 return equality_family::none;
47 }
48}();

93// The value domain admitted by equal's family gates: stable equality whose
94// comparison cannot throw (so the noexcept guarantee cannot be violated) and
95// free of standard-library range components at any depth (equal is not part of
96// the C++ interoperability surface).
97template <typename T>
98concept EqualDomainValue = concepts::NothrowEqualityComparable<T> && !contains_standard_range_v<T>;

100// Equality-family gates shared by equal's overloads and not_equal's delegated
101// gate (REQ-FN-002G, REQ-FN-002H); these are not ordering capabilities.
102template <typename Lhs, typename Rhs>
103concept EqualScalarPairAdmissible =
104 equal_family_of_v<Lhs> == equality_family::none && equal_family_of_v<Rhs> == equality_family::none &&
105 in_non_cljonic_fallthrough_domain_v<Lhs> && in_non_cljonic_fallthrough_domain_v<Rhs> && std::same_as<Lhs, Rhs> &&
106 EqualDomainValue<Lhs>;
107
108template <typename Lhs, typename Rhs>
109concept EqualSequentialPairAdmissible =
110 equal_family_of_v<Lhs> == equality_family::sequential && equal_family_of_v<Rhs> == equality_family::sequential &&
111 std::same_as<equal_value_type_of_t<Lhs>, equal_value_type_of_t<Rhs>> &&
112 EqualDomainValue<equal_value_type_of_t<Lhs>>;
113
114template <typename Lhs, typename Rhs>
115concept EqualStringPairAdmissible =
116 equal_family_of_v<Lhs> == equality_family::string && equal_family_of_v<Rhs> == equality_family::string;
117
118template <typename Lhs, typename Rhs>
119concept EqualMapPairAdmissible =
120 equal_family_of_v<Lhs> == equality_family::map && equal_family_of_v<Rhs> == equality_family::map &&
121 std::same_as<equal_key_type_of_t<Lhs>, equal_key_type_of_t<Rhs>> &&
122 std::same_as<equal_association_value_type_of_t<Lhs>, equal_association_value_type_of_t<Rhs>> &&
123 EqualDomainValue<equal_key_type_of_t<Lhs>> && EqualDomainValue<equal_association_value_type_of_t<Lhs>>;
124
125template <typename Lhs, typename Rhs>
126concept EqualSetPairAdmissible =
127 equal_family_of_v<Lhs> == equality_family::set && equal_family_of_v<Rhs> == equality_family::set &&
128 std::same_as<equal_value_type_of_t<Lhs>, equal_value_type_of_t<Rhs>> &&
129 EqualDomainValue<equal_value_type_of_t<Lhs>>;

131// Keep every arity on the same family gates so unsupported pairs fail at
132// compile time regardless of their position in the argument list.
133template <typename Lhs, typename Rhs>
134concept EqualPairAdmissible =
135 EqualScalarPairAdmissible<Lhs, Rhs> || EqualSequentialPairAdmissible<Lhs, Rhs> ||
136 EqualStringPairAdmissible<Lhs, Rhs> || EqualMapPairAdmissible<Lhs, Rhs> || EqualSetPairAdmissible<Lhs, Rhs>;

151// Lazy element-wise bounded-prefix walk: terminates at the first differing
152// element pair and requires both operands to be exhausted simultaneously for
153// equality. All producers expose cap-bounded begin()/end() traversal, so the
154// walk always terminates, including for two unbounded producers.
155namespace equal_walk_detail {
156
157// Reports whether both iterators have reached their respective ends after the
158// walk, which requires the two sequences to have exhausted simultaneously.
159template <typename LhsIterator, typename RhsIterator, typename LhsEnd, typename RhsEnd>
160[[nodiscard]] constexpr auto both_exhausted(const LhsIterator& lhs_iterator, const LhsEnd& lhs_end,
161 const RhsIterator& rhs_iterator, const RhsEnd& rhs_end) noexcept -> bool {
162 return (lhs_iterator == lhs_end) && (rhs_iterator == rhs_end);
163}
164
165} // namespace equal_walk_detail
166
167template <typename Lhs, typename Rhs>
168[[nodiscard]] constexpr auto equal_prefix_walk(const Lhs& lhs, const Rhs& rhs) noexcept -> bool {
169 using equal_walk_detail::both_exhausted;
170 auto lhs_iterator = lhs.begin();
171 const auto lhs_end = lhs.end();
172 auto rhs_iterator = rhs.begin();
173 const auto rhs_end = rhs.end();
174 while ((lhs_iterator != lhs_end) && (rhs_iterator != rhs_end)) {
175 if (!(*lhs_iterator == *rhs_iterator)) {
176 return false;
177 }
178 ++lhs_iterator;
179 ++rhs_iterator;
180 }
181 return both_exhausted(lhs_iterator, lhs_end, rhs_iterator, rhs_end);
182}

188// Order-insensitive Map matching: every logical entry of the left operand must
189// have a matching key in the right operand whose associated value compares
190// equal (REQ-COLL-021 Map equality semantics).
191template <typename Lhs, typename Rhs>
192[[nodiscard]] constexpr auto map_entries_all_match(const Lhs& lhs, const Rhs& rhs) noexcept -> bool {
193 for (const auto& entry : lhs) {
194 if (!rhs.contains(entry.key) || !(rhs(entry.key) == entry.value)) {
195 return false;
196 }
197 }
198 return true;
199}
200
201// Order-insensitive Set matching: every logical element of the left operand
202// must be present in the right operand (REQ-COLL-021 Set equality semantics).
203template <typename Lhs, typename Rhs>
204[[nodiscard]] constexpr auto set_elements_all_contained(const Lhs& lhs, const Rhs& rhs) noexcept -> bool {
205 for (const auto& element : lhs) {
206 if (!rhs.contains(element)) {
207 return false;
208 }
209 }
210 return true;
211}
