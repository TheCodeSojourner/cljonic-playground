60// A single (key, value) pair is admissible for the variadic form when the
61// collection admits it under its own key/value domain (identical to the
62// single-pair overload's gate).
63template <typename C, typename K, typename V>
64concept AssocPairAdmissible =
65 concepts::AssociativeCollection<C> && requires(const C& collection, const K& key, const V& value) {
66 { collection.assoc(key, value) } noexcept -> std::same_as<C>;
67 };

69// Every (key, value) pair in the argument list is admissible. The recursion
70// consumes two arguments per step; an empty tail is admissible and a single
71// trailing argument (a key without its value) is not, so an odd argument count
72// can never satisfy the constraint.
73template <typename C, typename... Pairs>
74struct all_assoc_pairs_admissible : std::false_type {};
75
76template <typename C>
77struct all_assoc_pairs_admissible<C> : std::true_type {};

79template <typename C, typename K, typename V, typename... Rest>
80struct all_assoc_pairs_admissible<C, K, V, Rest...>
81 : std::bool_constant<AssocPairAdmissible<C, K, V> && all_assoc_pairs_admissible<C, Rest...>::value> {};

83template <typename C, typename... Pairs>
84inline constexpr bool all_assoc_pairs_admissible_v = all_assoc_pairs_admissible<C, Pairs...>::value;

86// Left-to-right fold of the remaining pairs into an accumulator that already
87// holds the first pair(s); each step reuses the single-pair overload's contract.
88// The accumulator is threaded by reference (never passed by value) so the fold
89// cannot be intercepted by a collection's source-construction constructor.
90template <typename C>
91constexpr void fold_assoc_pairs_into([[maybe_unused]] C& result) noexcept {
92}
