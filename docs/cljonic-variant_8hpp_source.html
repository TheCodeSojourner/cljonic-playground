13// The converting constructor selects exactly one alternative by decayed type;
14// zero or multiple matches leave the constructor non-viable, so an ambiguous or
15// foreign argument is rejected at compile time rather than converting.
16template <typename T, typename... Alternatives>
17inline constexpr std::size_t matching_alternative_count_v =
18 (std::size_t{0} + ... + (std::same_as<T, Alternatives> ? std::size_t{1} : std::size_t{0}));

20// Alternative-strict equality within a single active index. Recursion is by
21// index so only same-alternative comparisons are instantiated: a two-variant
22// std::visit would instantiate cross-alternative comparisons (which need not
23// exist) and fail to compile.
24template <std::size_t Index = 0, typename Variant>
25[[nodiscard]] constexpr auto same_index_equal(const Variant& lhs, const Variant& rhs) noexcept -> bool {
26 if constexpr (Index + 1 == std::variant_size_v<Variant>) {
27 // Last alternative: the caller guarantees the active index matches.
28 return std::get<Index>(lhs) == std::get<Index>(rhs);
29 } else {
30 if (lhs.index() == Index) {
31 return std::get<Index>(lhs) == std::get<Index>(rhs);
32 }
33 return same_index_equal<Index + 1>(lhs, rhs);
34 }
35}
36
37// Alternative-strict ordering within a single active index; mirrors
38// same_index_equal so only same-alternative comparisons are instantiated.
39template <std::size_t Index = 0, typename Variant>
40[[nodiscard]] constexpr auto same_index_less(const Variant& lhs, const Variant& rhs) noexcept -> bool {
41 if constexpr (Index + 1 == std::variant_size_v<Variant>) {
42 // Last alternative: the caller guarantees the active index matches.
43 return std::get<Index>(lhs) < std::get<Index>(rhs);
44 } else {
45 if (lhs.index() == Index) {
46 return std::get<Index>(lhs) < std::get<Index>(rhs);
47 }
48 return same_index_less<Index + 1>(lhs, rhs);
49 }
50}
