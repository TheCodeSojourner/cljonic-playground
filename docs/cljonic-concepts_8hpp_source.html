27// Convert an integral API index only when its value is representable by the
28// library's normalized size_t index domain. Collection-specific bounds remain
29// at each collection's call site because lookup and association have different
30// valid ranges.
31template <std::integral IndexType>
32[[nodiscard]] constexpr auto try_normalize_index(IndexType index) noexcept -> std::optional<std::size_t> {
33 if (!std::in_range<std::size_t>(index)) {
34 return std::nullopt;
35 }
36 return static_cast<std::size_t>(index);
37}

72// Static sources must fit their destination capacity before construction;
73// dynamic sources defer that decision to the bounded copy loop.
74template <typename T, std::size_t CapacityValue>

96// Recursive component analysis for composite values in the closed cljonic
97// value domain (REQ-CAP-010). `contains_floating_point_v<T>` reports whether a
98// floating-point type occurs as T itself or inside any stored component of a
99// supported composite (std::variant alternatives, cljonic collections, and
100// MapEntry). `contains_callable_v<T>` reports whether a callable type occurs as
101// a component of a composite (including function pointers); callables never
102// admit stable value equality.
103template <typename T>
104struct contains_floating_point : std::bool_constant<std::floating_point<std::remove_cvref_t<T>>> {};
105
106template <typename T>
107inline constexpr bool contains_floating_point_v = contains_floating_point<std::remove_cvref_t<T>>::value;
108
109template <typename T>
110struct contains_callable : std::false_type {};
111
112template <typename T>
113inline constexpr bool contains_callable_v = contains_callable<std::remove_cvref_t<T>>::value;
114
115// A function pointer is callable even though it happens to define operator==;
116// the callable-component rejection rule treats it as a callable for composite
117// admission (REQ-CAP-010) so address-as-key semantics are never exposed.
118template <typename Return, typename... Args>
119struct contains_callable<Return (*)(Args...)> : std::true_type {};
120
121template <typename Return, typename... Args>
122struct contains_callable<Return (*)(Args...) noexcept> : std::true_type {};
123
124// std::variant: recurse into every alternative.
125template <typename... Alternatives>
126struct contains_floating_point<std::variant<Alternatives...>>
127 : std::bool_constant<(contains_floating_point_v<Alternatives> || ...)> {};
128
129template <typename... Alternatives>
130struct contains_callable<std::variant<Alternatives...>>
131 : std::bool_constant<(contains_callable_v<Alternatives> || ...)> {};
