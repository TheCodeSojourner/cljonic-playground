20// A small closed-world tag lets concepts distinguish collection families
21// without exposing implementation-specific type traits as public API.
22enum class collection_kind { none, vector, map, set, queue, string };

24// Convert an integral API index only when its value is representable by the
25// library's normalized size_t index domain. Collection-specific bounds remain
26// at each collection's call site because lookup and association have different
27// valid ranges.
28template <std::integral IndexType>
29[[nodiscard]] constexpr auto try_normalize_index(IndexType index) noexcept -> std::optional<std::size_t> {
30 if (!std::in_range<std::size_t>(index)) {
31 return std::nullopt;
32 }
33 return static_cast<std::size_t>(index);
34}

52// Extract a source's compile-time element count when one is knowable. A
53// dynamic extent remains acceptable because it must be checked at runtime.
54template <typename T>
55struct static_extent : std::integral_constant<std::size_t, std::dynamic_extent> {};
56
57template <typename ElementType, std::size_t Extent>
58struct static_extent<std::span<ElementType, Extent>> : std::integral_constant<std::size_t, Extent> {};
59
60template <typename ElementType, std::size_t Extent>
61struct static_extent<std::array<ElementType, Extent>> : std::integral_constant<std::size_t, Extent> {};
62
63template <typename ElementType, std::size_t Extent>
64struct static_extent<ElementType[Extent]> : std::integral_constant<std::size_t, Extent> {};
65
66template <typename T>
67inline constexpr std::size_t static_extent_v = static_extent<std::remove_cvref_t<T>>::value;
68
69// Static sources must fit their destination capacity before construction;
70// dynamic sources defer that decision to the bounded copy loop.
71template <typename T, std::size_t CapacityValue>
72inline constexpr bool static_extent_fits_v =
73 static_extent_v<T> == std::dynamic_extent || static_extent_v<T> <= CapacityValue;
