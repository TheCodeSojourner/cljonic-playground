21// A small closed-world tag lets concepts distinguish collection families
22// without exposing implementation-specific type traits as public API.
23enum class collection_kind { none, vector, map, set, queue, string };
24
25// Convert an integral API index only when its value is representable by the
26// library's normalized size_t index domain. Collection-specific bounds remain
27// at each collection's call site because lookup and association have different
28// valid ranges.
29template <std::integral IndexType>
30[[nodiscard]] constexpr auto try_normalize_index(IndexType index) noexcept -> std::optional<std::size_t> {
31 if (!std::in_range<std::size_t>(index)) {
32 return std::nullopt;
33 }
34 return static_cast<std::size_t>(index);
35}

53// Extract a source's compile-time element count when one is knowable. A
54// dynamic extent remains acceptable because it must be checked at runtime.
55template <typename T>
56struct static_extent : std::integral_constant<std::size_t, std::dynamic_extent> {};
57
58template <typename ElementType, std::size_t Extent>
59struct static_extent<std::span<ElementType, Extent>> : std::integral_constant<std::size_t, Extent> {};
60
61template <typename ElementType, std::size_t Extent>
62struct static_extent<std::array<ElementType, Extent>> : std::integral_constant<std::size_t, Extent> {};
63
64template <typename ElementType, std::size_t Extent>
65struct static_extent<ElementType[Extent]> : std::integral_constant<std::size_t, Extent> {};
66
67template <typename T>
68inline constexpr std::size_t static_extent_v = static_extent<std::remove_cvref_t<T>>::value;
69
70// Static sources must fit their destination capacity before construction;
71// dynamic sources defer that decision to the bounded copy loop.
72template <typename T, std::size_t CapacityValue>
73inline constexpr bool static_extent_fits_v =
74 static_extent_v<T> == std::dynamic_extent || static_extent_v<T> <= CapacityValue;
