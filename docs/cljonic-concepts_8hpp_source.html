133// Convert an admitted API index only when its value is representable by the
134// library's normalized size_t index domain. Collection-specific bounds remain
135// at each collection's call site because lookup and association have different
136// valid ranges.
137template <IntegerIndexType IndexType>
138[[nodiscard]] constexpr auto try_normalize_index(IndexType index) noexcept -> std::optional<std::size_t> {
139 if (!std::in_range<std::size_t>(index)) {
140 return std::nullopt;
141 }
142 return static_cast<std::size_t>(index);
143}

254// Non-throwing equality of a type's own comparison (REQ-FN-002G): the type's own
255// operator== must be declared noexcept. This is the recursive guarantee behind
256// operations declared noexcept; a cljonic::Variant declares its own noexcept
257// ==, so detection succeeds through that operator.
258template <typename T>
259inline constexpr bool nothrow_equality_v = requires(const T& left, const T& right) {
260 { left == right } noexcept -> std::convertible_to<bool>;

263// A standard-library range or container is outside the equality domain: `equal`
