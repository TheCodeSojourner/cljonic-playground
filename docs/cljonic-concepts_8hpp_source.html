155// Convert an admitted API index only when its value is representable by the
156// library's normalized size_t index domain. Collection-specific bounds remain
157// at each collection's call site because lookup and association have different
158// valid ranges.
159template <IntegerIndexType IndexType>
160[[nodiscard]] constexpr auto try_normalize_index(IndexType index) noexcept -> std::optional<std::size_t> {
161 if (!std::in_range<std::size_t>(index)) {
162 return std::nullopt;
163 }
164 return static_cast<std::size_t>(index);

276// Non-throwing equality of a type's own comparison (REQ-FN-002G): the type's own
277// operator== must be declared noexcept. This is the recursive guarantee behind
278// operations declared noexcept; a cljonic::Variant declares its own noexcept
279// ==, so detection succeeds through that operator.
280template <typename T>
281inline constexpr bool nothrow_equality_v = requires(const T& left, const T& right) {
282 { left == right } noexcept -> std::convertible_to<bool>;

285// A standard-library range or container is outside the equality domain: `equal`
