147 // Copy non-cljonic character ranges into owned storage, retaining the
148 // bounded prefix for dynamic extents; statically known extents must fit.
149 // A cljonic collection or producer is not a SourceConstruction source
150 // (REQ-FN-027A); materialize it via into, preflighted by fits_into.
151 template <std::ranges::input_range SourceRange>
152 requires(!concepts_detail::is_cljonic_collection_v<SourceRange> &&
153 !concepts_detail::is_cljonic_producer_v<SourceRange> &&
154 !std::same_as<std::remove_cvref_t<SourceRange>, std::string_view> &&
155 std::same_as<std::remove_cvref_t<std::ranges::range_reference_t<SourceRange>>, char>)
156 explicit constexpr String(SourceRange&& source) noexcept {
157 static_assert(concepts::NothrowRangeConstruction<char, SourceRange>,
158 "cljonic::String: range/view constructor requires char elements and non-throwing traversal");
159 static_assert(concepts_detail::static_extent_fits_v<SourceRange, CapacityValue>,
160 "cljonic::String: static-extent range source exceeds String CapacityValue");
161
162 std::size_t copy_count = 0;
163 for (auto&& byte : std::forward<SourceRange>(source)) {
164 if (copy_count >= CapacityValue) {
165 break;
166 }
167 data_[copy_count++] = normalize_byte(static_cast<char>(byte));
168 }
169 logical_size_ = copy_count;
170 data_[logical_size_] = '\0';
171 }

367template <typename Source, typename... Args>
368constexpr auto string_factory_reject_text_source() noexcept -> void {
369 if constexpr (is_cljonic_collection_v<Source> || is_cljonic_producer_v<Source>) {
370 static_assert(dependent_false<Args...>,
371 "cljonic::string: a cljonic collection or producer is not a text source. A String is built "
372 "from a string literal, or from a non-cljonic C++ range or view with an explicit capacity.");
373 } else if constexpr (std::same_as<Source, std::string_view>) {
374 static_assert(dependent_false<Args...>,
375 "cljonic::string: a string_view's size is not part of its type; provide an explicit "
376 "capacity: string<Capacity>(view).");
377 } else {
378 static_assert(dependent_false<Args...>,
379 "cljonic::string: the argument must be a string literal, or with an explicit capacity a "
380 "string_view or non-cljonic char range or view.");
381 }
