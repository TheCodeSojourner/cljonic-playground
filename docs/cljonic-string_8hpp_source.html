139 // Copy non-cljonic character ranges into owned storage, retaining the
140 // bounded prefix for dynamic extents; statically known extents must fit.
141 // A cljonic collection or producer is not a SourceConstruction source
142 // (REQ-FN-027A); materialize it via into, preflighted by fits_into.
143 template <std::ranges::input_range SourceRange>
144 requires(!concepts_detail::is_cljonic_collection_v<SourceRange> &&
145 !concepts_detail::is_cljonic_producer_v<SourceRange> &&
146 !std::same_as<std::remove_cvref_t<SourceRange>, std::string_view> &&
147 std::same_as<std::remove_cvref_t<std::ranges::range_reference_t<SourceRange>>, char>)
148 constexpr String(SourceRange&& source) noexcept {
149 static_assert(concepts::NothrowRangeConstruction<char, SourceRange>,
150 "String range/view constructor requires char elements and non-throwing traversal");
151 static_assert(concepts_detail::static_extent_fits_v<SourceRange, CapacityValue>,
152 "String static-extent range source exceeds String CapacityValue");
153
154 std::size_t copy_count = 0;
155 for (auto&& byte : std::forward<SourceRange>(source)) {
156 if (copy_count >= CapacityValue) {
157 break;
158 }
159 data_[copy_count++] = normalize_byte(static_cast<char>(byte));
160 }
161 logical_size_ = copy_count;
162 data_[logical_size_] = '\0';
163 }

165 // Diagnostic fallback (REQ-DIAG-010): a cljonic collection or producer is
166 // not a SourceConstruction source. This overload exists only to explain the
167 // rejection -- it produces no value and is never a supported call target.
168 template <typename SourceValue>
169 requires((concepts_detail::is_cljonic_collection_v<SourceValue> ||
170 concepts_detail::is_cljonic_producer_v<SourceValue>) &&
171 !std::same_as<std::remove_cvref_t<SourceValue>, value_type>)
172 constexpr String([[maybe_unused]] SourceValue&& source) noexcept {
173 static_assert(concepts_detail::dependent_false<SourceValue>,
174 "cljonic::String: you cannot construct a String from a cljonic collection or producer. A String "
175 "is built from its characters, or from a non-cljonic C++ range or view. To copy this argument's "
176 "contents into a String, use into(destination, source); call fits_into(destination, source) "
177 "first to check whether the whole source fits.");
178 }
